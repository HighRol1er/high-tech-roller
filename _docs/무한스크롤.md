```ts
// action/post.ts
'use server';

import { supabase } from '@/lib/supabase';

export async function fetchMorePosts(page: number, pageSize: number, tag?: string) {
  const from = page * pageSize;
  const to = from + pageSize - 1;

  let query = supabase
    .from('posts')
    .select('id, title, tags, slug, created_at')
    .order('created_at', { ascending: false })
    .range(from, to);

  if (tag) {
    query = query.contains('tags', [tag]);
  }

  const { data, error } = await query;
  if (error) throw error;
  return data ?? [];
}
```

```tsx
// app/page.tsx
const PAGE_SIZE = 12;

const getInitialPosts = async () => {
  try {
    const { data, error } = await supabase
      .from('posts')
      .select('id, title, tags, slug, created_at')
      .order('created_at', { ascending: false })
      .range(0, PAGE_SIZE - 1);

    if (error) throw error;
    return data ?? [];
  } catch (error) {
    console.error('error', error);
    return [];
  }
};

export default async function Main() {
  const allPosts = await getInitialPosts();

  return (
    <div className='max-w-7xl mx-auto'>
      <Suspense fallback={<Loader />}>
        <PostList initialPosts={allPosts} pageSize={PAGE_SIZE} />
      </Suspense>
    </div>
  );
}
```

```tsx
// postList.tsx
'use client';

import type { PostSummary } from '@/types';

import { PostCard } from '@/components/post-card/PostCard';
import { slugify } from '@/lib';
import { useSearchParams } from 'next/navigation';
import { fetchMorePosts } from '@/app/actions/posts';
import { useEffect, useRef, useState } from 'react';
import { Loader } from '../common';

interface PostListProps {
  initialPosts: PostSummary[];
  pageSize: number;
}

export const PostList = ({ initialPosts, pageSize }: PostListProps) => {
  const searchParams = useSearchParams();
  const tagParam = searchParams.get('tag');

  const [posts, setPosts] = useState<PostSummary[]>(initialPosts);
  const [page, setPage] = useState(1);
  const [hasMore, setHasMore] = useState(initialPosts.length === pageSize);
  const [isLoading, setIsLoading] = useState(false);
  const observerRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    setPosts(initialPosts);
    setPage(1);
    setHasMore(initialPosts.length === pageSize);
  }, [tagParam, initialPosts, pageSize]);

  useEffect(() => {
    const observer = new IntersectionObserver(
      async (entries) => {
        if (!entries[0].isIntersecting || isLoading || !hasMore) return;

        setIsLoading(true);
        try {
          const newPosts = await fetchMorePosts(page, pageSize, tagParam ?? undefined);
          if (newPosts.length < pageSize) setHasMore(false);
          setPosts((prev) => [...prev, ...newPosts]);
          setPage((prev) => prev + 1);
        } catch (error) {
          console.error(error);
        } finally {
          setIsLoading(false);
        }
      },
      { threshold: 0.5 },
    );

    if (observerRef.current) observer.observe(observerRef.current);
    return () => observer.disconnect();
  }, [page, isLoading, hasMore, pageSize, tagParam]);

  const filteredPosts = tagParam
    ? posts.filter((p) => p.tags.some((tag) => slugify(tag) === tagParam.toLowerCase()))
    : posts;

  return (
    <div>
      <div className='grid grid-cols-1 sm:grid-cols-2 md:grid-cols-3 lg:grid-cols-4 gap-4 p-4'>
        {filteredPosts.map((post) => (
          <PostCard key={post.id} post={post} />
        ))}
      </div>

      <div ref={observerRef} className='h-10' />

      {isLoading && <Loader />}
      {!hasMore && <p className='text-center text-sm text-muted-foreground py-8'>모든 게시글을 불러왔어요.</p>}
    </div>
  );
};
```
